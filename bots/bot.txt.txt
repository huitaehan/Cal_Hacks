void move(int ma, int mb, int mc, int md)
{
  //PWM scale convert 0~255 to 0~100(approx)
  int va = ma2.55;
  int vb = mb2.55;
  int vc = mc2.55;
  int vd = md2.55;

  MOTORA(va);
  MOTORB(vc);
  MOTORC(vb);
  //MOTORD(vd);
}

void MOTORA(int ma)
{
  int MA = abs(ma);
  if(ma < 0)
  {
    analogWrite(2,0);
    analogWrite(3,MA);
  }
  else if(ma > 0)
  {
    analogWrite(2,ma);
    analogWrite(3,0);
  }
  else
  {
    analogWrite(2,0);
    analogWrite(3,0);
  }
}

void MOTORB(int mb)
{
int MB = abs(mb);
  if(mb < 0)
  {
    analogWrite(0,0);
    analogWrite(1,MB);
  }
  else if(mb > 0)
  {
    analogWrite(0,mb);
    analogWrite(1,0);
  }
  else
  {
    analogWrite(0,0);
    analogWrite(1,0);
  }
}
void MOTORC(int mc)
{
int MC = abs(mc);
  if(mc < 0)
  {
    analogWrite(8,0);
    analogWrite(9,MC);
  }
  else if(mc > 0)
  {
    analogWrite(8,mc);
    analogWrite(9,0);
  }
  else
  {
    analogWrite(8,0);
    analogWrite(9,0);
  }
}

void dir_move(int dir, int vel)
{
  double Alpha = dir * 30 * PI / 180;
  double ax = vel * cos(Alpha);
  double ay = vel * sin(Alpha);
  double w = 1;//1.5 * (heading-180);

  int F1 = (int)(0.578 * ax - 0.333 * ay + 0.333 * w);
  int F2 = (int)(0 * ax + 0.567 * ay + 0.333 * w);
  int F3 = (int)(-0.578 * ax - 0.333 * ay + 0.333 * w);

  int maxF = max(abs(F1), max(abs(F2), abs(F3)));

  if (maxF > 0) {
    F1 = F1 * vel / maxF;
    F2 = F2 * vel / maxF;
    F3 = F3 * vel / maxF;
  }

  move(F1,F2,F3, 0);

  delay(20);
}
